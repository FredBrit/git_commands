# Скачивает репозиторий с GitHub на твой компьютер. Создаёт папку с проектом.
  git clone https://github.com/FredBrit/fluent_python.git

# Переходит в папку с репозиторием (чтобы работать внутри неё).
  cd fluent_python

# Показывает:
# Какие файлы изменились
# Какие файлы новые (не отслеживаются)
# Готовы ли изменения к коммиту
  git status

# Добавляет файл в "индекс" (подготавливает к сохранению). Git начинает отслеживать этот файл.
  git add test.ipynb
  git add . — добавить все изменённые файлы
  git add *.ipynb — добавить все notebook файлы

# Сохраняет изменения локально на твоём компьютере. Создаёт "снимок" (snapshot) текущей версии файлов. Сообщение должно кратко описывать, что изменилось.
  git commit -m "Add test notebook"

# Зачем: Переименовывает текущую ветку в main.
# Контекст: Раньше основная ветка в Git называлась master, но сейчас многие репозитории (включая GitHub) используют название main.
  git branch -M main

# Отправляет все локальные коммиты на GitHub. После этого изменения видны в репозитории на сайте.
  git push

# Это команда первой отправки новой ветки на сервер.
# git push -u origin main

# Скачивает изменения с GitHub и пытается автоматически объединить (merge) их с вашими локальными изменениями.
# git pull

# Переименовывает файл (это команда Windows, не Git).
  ren Untitled.ipynb match_case.ipynb

Более сложные команды

# git checkout --ours middle_tasks.ipynb или git checkout --theirs middle_tasks.ipynb (при конфликте версий для слияния можно выбрать либо локальную (ours), либо с гитхаба (theirs))
# git add middle_tasks.ipynb

# Скачивает изменения с GitHub и аккуратно накладывает ваши локальные коммиты поверх них.
# git pull --rebase

# Полностью отменяет процесс rebase и возвращает репозиторий в состояние ровно до начала команды.
# git rebase --abort

# Полностью отменяет процесс слияния (merge) и возвращает всё как было.
# git merge --abort

# Временно "прячет" все несохранённые изменения в безопасное хранилище Git, делая рабочую папку идеально чистой.
# git stash

# Достает спрятанные изменения из хранилища и применяет их обратно к файлам.
# git stash pop
