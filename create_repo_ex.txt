# Создание репозитория на примере Airflow

# 1. Создай репозиторий на GitHub (веб-интерфейс)
#    Name: airflow-deploy
#    Visibility: Private (рекомендуется, там будут секреты)
#    НЕ ставь галочки "Add README", ".gitignore", "License"

# 2. Клонируй пустой репозиторий локально
git clone https://github.com/ТВОЙ_НИК/airflow-deploy.git
cd airflow-deploy

# 3. Создай структуру папок и файлы
mkdir -p config/postgres scripts dags plugins logs

# Создай пустые файлы для Git (чтобы папки не игнорировались)
touch dags/.gitkeep plugins/.gitkeep logs/.gitkeep

# 4. Создай/вставь файлы (можно через VS Code, nano, или cat << 'EOF')
#    Например:
nano .env.example        # вставь содержимое
nano .gitignore          # вставь содержимое
nano docker-compose.yaml # вставь содержимое
nano Dockerfile          # вставь содержимое
nano config/airflow.cfg  # вставь содержимое
nano config/postgres/postgresql.conf # вставь содержимое
nano scripts/deploy.sh   # вставь содержимое
nano README.md           # вставь содержимое

# 5. Дай права на выполнение скрипту
chmod +x scripts/deploy.sh

# 6. Закоммить и отправь на GitHub
git add .
git commit -m "feat: init airflow 3.x deploy stack"
git branch -M main
git push -u origin main


# Далее склонировать репозиторий
git clone https://github.com/ТВОЙ_НИК/airflow-deploy.git /opt/airflow

# Перейти в директорию, накинуть права и запустить скрипт инициализации, если он есть
cd /opt/airflow && chmod +x scripts/deploy.sh && ./scripts/deploy.sh

### Создание проекта в гите на основе уже существующего проекта на машине

## Сохранить зависимости в файл requirements.txt
 pip freeze > requirements.txt

## Создать файл .gitignore в корне проекта и добавить туда следующее

# Виртуальное окружение
 venv/
 .venv/
 env/

 # Python
 __pycache__/
 *.py[cod]
 *$py.class
 *.so

 # IDE
 .vscode/
 .idea/
 *.swp
 *.swo

 # Переменные окружения (никогда не коммитьте пароли!)
 .env


## Создай репозиторий на GitHub (веб-интерфейс)
#    Name: example
#    Visibility: Private/Public
#    НЕ ставь галочки "Add README", ".gitignore", "License"

## Выполняем последовательно команды
# 1. Инициализируем Git в текущей папке
git init

# 2. Добавляем все файлы (кроме тех, что в .gitignore, включая venv!)
git add .

# 3. Делаем первый коммит
git commit -m "Initial commit: FastAPI project structure"

# 4. Переименовываем главную ветку в main (современный стандарт)
git branch -M main

# 5. Привязываем локальный репозиторий к вашему GitHub (вставьте свой URL)
git remote add origin https://github.com/ВАШ_ЛОГИН/ВАШ_РЕПОЗИТОРИЙ.git

# 6. Отправляем код на GitHub
git push -u origin main

## Запуск на другой машине

# 1. Клонируем репозиторий
git clone https://github.com/ВАШ_ЛОГИН/ВАШ_РЕПОЗИТОРИЙ.git
cd ВАШ_РЕПОЗИТОРИЙ

# 2. Создаем НОВОЕ виртуальное окружение под эту систему
python3 -m venv venv

# 3. Активируем его
source venv/bin/activate  # (для Windows: venv\Scripts\activate)

# 4. Устанавливаем все зависимости из файла
pip install -r requirements.txt

# 5. Запускаем проект
python3 main.py
