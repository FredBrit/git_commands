### Подключение локального репозитория к GitLab
# Проверить права на ключ (SSH отказывается работать с слишком открытыми правами)
chmod 600 ~/.ssh/id_ed25519 && chmod 644 ~/.ssh/id_ed25519.pub

# Вывести публичный ключ — скопируйте всю строку и вставьте в GitLab → Settings → SSH Keys
cat ~/.ssh/id_ed25519.pub

# Проверить, что GitLab принимает ключ (должно ответить: Welcome to GitLab, @FredBrit!)
ssh -T git@gitlab.com

# Перейти в папку проекта
cd /opt/edu_fast_api

# Добавить GitLab как второй remote (origin уже занят GitHub)
git remote add gitlab git@gitlab.com:FredBrit/booking.git

# Убедиться, что оба remote на месте
git remote -v

# Запушить текущую ветку в GitLab и запомнить связь
git push -u gitlab main

# Дальнейшая работа: пуш в GitHub
git push origin main

# Дальнейшая работа: пуш в GitLab
git push gitlab main







### Подключение целевой машины к GitLab для развертывания проекта через GitLab CI и Registry

# 1. Установить Docker
curl -fsSL https://get.docker.com | sh

# 2. Docker Compose уже идёт в составе Docker (проверить)
docker compose version

# 3. Добавить пользователя в группу docker (чтобы не писать sudo)
usermod -aG docker $USER
newgrp docker

# 4. Создать папку для проекта
mkdir -p /opt/booking
cd /opt/booking

# 5. Создать docker-compose.yml и все переменные окружения

# 6. Вставь публичный SSH-ключ (который скопировал с локальной машины)
# На локальной: 
cat gitlab_deploy.pub

# На целевой машине:
mkdir -p ~/.ssh
chmod 700 ~/.ssh
nano ~/.ssh/authorized_keys
# Вставь строку из gitlab_deploy.pub
# Сохрани: Ctrl+O, Enter, выход: Ctrl+X
chmod 600 ~/.ssh/authorized_keys
