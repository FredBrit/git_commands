### Подключение локального репозитория к GitLab
# Проверить права на ключ (SSH отказывается работать с слишком открытыми правами)
chmod 600 ~/.ssh/id_ed25519 && chmod 644 ~/.ssh/id_ed25519.pub

# Вывести публичный ключ — скопируйте всю строку и вставьте в GitLab → Settings → SSH Keys
cat ~/.ssh/id_ed25519.pub

# Проверить, что GitLab принимает ключ (должно ответить: Welcome to GitLab, @FredBrit!)
ssh -T git@gitlab.com

# Перейти в папку проекта
cd /opt/edu_fast_api

# Добавить GitLab как второй remote (origin уже занят GitHub)
git remote add gitlab git@gitlab.com:FredBrit/booking.git

# Убедиться, что оба remote на месте
git remote -v

# Запушить текущую ветку в GitLab и запомнить связь
git push -u gitlab main

# Дальнейшая работа: пуш в GitHub
git push origin main

# Дальнейшая работа: пуш в GitLab
git push gitlab main

### Подключение целевой машины к GitLab для развертывания проекта

# Установить Docker
curl -fsSL  -o get-docker.sh && sh get-docker.sh

# Установить плагин Docker Compose (входит в состав Docker CLI)
apt-get update && apt-get install -y docker-compose-plugin

# Проверить, что Docker и Compose работают
docker --version && docker compose version

# Создать папку для SSH-ключей
mkdir -p ~/.ssh && chmod 700 ~/.ssh

# Сгенерировать SSH-ключ (без passphrase, чтобы не ломать CI/CD)
ssh-keygen -t ed25519 -f ~/.ssh/id_ed25519 -N ""

# Вывести публичный ключ — скопируйте всю строку и вставьте в GitLab → Settings → SSH Keys
cat ~/.ssh/id_ed25519.pub

# Проверить, что GitLab принимает ключ (должно ответить: Welcome to GitLab, @FredBrit!)
ssh -T 

# Перейти в папку, куда будете клонировать проект
cd /opt

# Клонировать репозиторий из GitLab
git clone :FredBrit/booking.git

# Перейти в папку проекта
cd booking

# Установить Python и venv
apt-get install -y python3 python3-venv

# Создать виртуальное окружение
python3 -m venv venv

# Активировать виртуальное окружение
source venv/bin/activate

# Установить зависимости проекта
pip install -r requirements.txt

# Создать файл .env из примера (если есть .env.example)
cp .env.example .env

# Поднять проект через Docker Compose
docker compose up -d --build
